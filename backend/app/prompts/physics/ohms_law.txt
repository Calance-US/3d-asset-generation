Example code snippet for voice narration setup and control:

```js
// Initialize speech synthesis
const synth = window.speechSynthesis;
let utteranceQueue = [];

// Function to speak a given text
function speak(text) {
  if (!synth) return;
  const utterance = new SpeechSynthesisUtterance(text);
  utterance.lang = 'en-US'; // set language
  utterance.rate = 1;       // normal speed
  utterance.pitch = 1;      // normal pitch
  synth.speak(utterance);
  return utterance;
}

// Queue multiple narration segments sequentially
function speakSequence(textArray) {
  if (!synth) return;
  utteranceQueue = textArray.map(t => new SpeechSynthesisUtterance(t));
  utteranceQueue.forEach((u, i) => {
    u.lang = 'en-US';
    u.rate = 1;
    u.pitch = 1;
    if (i > 0) {
      utteranceQueue[i - 1].addEventListener('end', () => synth.speak(u));
    }
  });
  if (utteranceQueue.length > 0) synth.speak(utteranceQueue[0]);
}

// Example usage: play narration on scene start or user action
const narrationTexts = [
  "Welcome to the Ohm's Law simulation.",
  "Observe the relationship between voltage, current, and resistance.",
  "Toggle the switch to start or stop the current flow."
];
speakSequence(narrationTexts);

// Controls for play, pause, resume, and cancel
function pauseNarration() { if (synth.speaking) synth.pause(); }
function resumeNarration() { if (synth.paused) synth.resume(); }
function cancelNarration() { if (synth.speaking || synth.paused) synth.cancel(); }
```

* Synchronize narration segments with key animation states or user interactions by invoking `speak()` or `speakSequence()` accordingly.
